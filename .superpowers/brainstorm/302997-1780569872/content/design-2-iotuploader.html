🔽 命令解析逻辑

messageReceived 槽函数
void IotUploader::onMessageReceived(topic, payload) {
    // 1. 解析 JSON
    QJsonObject msg = QJsonDocument::fromJson(payload).object();
    QString serviceId = msg["service_id"].toString();
    QString cmdName = msg["command_name"].toString();
    QString requestId = msg["request_id"].toString();
    QJsonObject paras = msg["paras"].toObject();

    // 2. 识别命令
    if (serviceId == "pump_control") {
        bool on = paras["pump"].toBool();
        emit deviceCommandReceived("pump", on, requestId);
    } else if (serviceId == "light_control") {
        bool on = paras["light"].toBool();
        emit deviceCommandReceived("lamp", on, requestId);
    }
}

📤 响应方法

TeaGardenSystem 调用此方法发响应到云端
void IotUploader::sendCommandResponse(
    const QString &cmdName, // "pump_control" / "light"
    const QString &requestId,
    int result // 0=成功 1=失败
) {
    QString topic = QString(
        "$oc/devices/%1/sys/commands/response/request_id=%2"
    ).arg(deviceId, requestId);

    QJsonObject body;
    body["result_code"] = 0; // 平台通信成功
    body["response_name"] = cmdName;
    QJsonObject r;
    r["result"] = result;
    body["paras"] = r;

    m_client->publish(topic, QJsonDocument(body).toJson(QJsonDocument::Compact), 1);
}

// 新槽函数
void TeaGardenSystem::onIotCommand(
    device, on, requestId) {
    int cmd = (device == "pump") ? (on ? 3 : 4) : (on ? 1 : 2);
    bool ok = loraManager->sendCommand(cmd);
    int result = ok ? 0 : 1;
    QString cmdName = (device == "pump")
        ? "pump_control" : "light_control";
    IotUploader::instance()->sendCommandResponse(cmdName, requestId, result);
}
