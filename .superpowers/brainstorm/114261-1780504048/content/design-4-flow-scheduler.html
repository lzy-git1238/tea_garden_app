⏰ 调度引擎 — 核心流程

每分钟触发一次 checkSchedules()
checkSchedules() {
    now = QDateTime::currentDateTime()
    today_dow = now.dayOfWeek() // 1=周一
    today_time = now.toString("HH:mm")

    // 1. 检查到期的关闭任务
    for each (active_job in activeJobs) {
        if (now >= active_job.end_time) {
            sendLoraCommand(device, off_cmd)
            logToDb(action="off", volume=cumulative)
            hideFlowCard()
            activeJobs.remove(device)
        }
    }

    // 2. 检查到期的开启任务
    plans = ScheduleStore::queryEnabled(device, today_dow, today_time)
    for each (plan in plans) {
        if (!activeJobs.contains(device)) {
            sendLoraCommand(device, on_cmd)
            logToDb(action="on")
            activeJobs[device] = {end: now+plan.duration}
            showFlowCard(plan) // 仅 pump
        }
    }
}

🔄 流量卡片刷新 — 每秒更新

refreshFlowCard() {
    flow_rate = QSettings("pump/flow_rate", 2.5)
    elapsed_sec = activeJob.start_time.secsTo(now)
    current_volume = flow_rate * elapsed_sec / 60.0
    expected_volume = flow_rate * plan.duration_min
    
    // 更新卡片：已灌溉量、进度条、剩余时间
    flowCard->update(current_volume, expected_volume, remaining_sec)
}
